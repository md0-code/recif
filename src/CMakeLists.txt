cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" RECIF_VERSION)
project(Recif VERSION ${RECIF_VERSION} LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

option(RECIF_DESKTOP "Build the Windows control panel and SDL2 DAF terminal" ON)
configure_file(version.h.in generated/rc_version.h @ONLY)

if(WIN32)
    enable_language(RC)
    set(RECIF_ICON "${CMAKE_CURRENT_SOURCE_DIR}/../assets/recif.ico")
    set(RECIF_ICON_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/generated/recif_icon.rc")
    file(CONFIGURE OUTPUT "${RECIF_ICON_RESOURCE}" CONTENT "1 ICON \"${RECIF_ICON}\"\n")
    set_source_files_properties("${RECIF_ICON_RESOURCE}" PROPERTIES OBJECT_DEPENDS "${RECIF_ICON}")
endif()

function(recif_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:/W4;/utf-8>")
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Wpedantic>")
    endif()
endfunction()

add_library(recif_serial STATIC serial/rs232.c)
target_include_directories(recif_serial PUBLIC serial)
recif_warnings(recif_serial)

add_library(recif_core STATIC core/bru.c core/coral.c core/fp11.c core/guest.c core/machine.c core/pdp11.c
    core/tape.c hw/rh11.c hw/rp11.c hw/tm11.c)
target_include_directories(recif_core PUBLIC core hw)
target_link_libraries(recif_core PUBLIC recif_serial)
recif_warnings(recif_core)

add_library(recif_daf STATIC hw/daf2020.c hw/dafserial.c hw/font5x7.c hw/font8x8.c)
target_include_directories(recif_daf PUBLIC hw)
target_link_libraries(recif_daf PUBLIC recif_serial)
recif_warnings(recif_daf)

add_library(recif_transport STATIC host/serial_link.c)
target_include_directories(recif_transport PUBLIC host)
target_link_libraries(recif_transport PUBLIC recif_serial)
if(WIN32)
    target_sources(recif_transport PRIVATE host/stream_win32.c)
    target_link_libraries(recif_transport PRIVATE ws2_32)
else()
    target_sources(recif_transport PRIVATE host/stream_posix.c)
endif()
recif_warnings(recif_transport)

add_executable(recif app/cli.c app/machine_session.c app/prefs.c)
target_link_libraries(recif PRIVATE recif_core recif_transport)
target_include_directories(recif PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
recif_warnings(recif)

if(WIN32)
    target_sources(recif PRIVATE "${RECIF_ICON_RESOURCE}")
endif()
if(WIN32 AND MSVC)
    target_sources(recif PRIVATE app/recif.manifest)
endif()

if(RECIF_DESKTOP)

    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL2
        URL https://www.libsdl.org/release/SDL2-2.32.10.tar.gz
        URL_HASH SHA256=5f5993c530f084535c65a6879e9b26ad441169b3e25d789d83287040a9ca5165
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(SDL2)

    add_library(recif_platform STATIC platform/video.c platform/png.c)
    target_include_directories(recif_platform PUBLIC platform)
    target_link_libraries(recif_platform PUBLIC SDL2::SDL2-static)
    target_compile_definitions(recif_platform PUBLIC SDL_MAIN_HANDLED)
    recif_warnings(recif_platform)
    if(NOT MSVC)
        # The OpenGL loader casts SDL_GL_GetProcAddress results; the CRT shader is one long string.
        target_compile_options(recif_platform PRIVATE -Wno-pedantic -Wno-overlength-strings)
    endif()

    target_compile_definitions(recif PRIVATE RECIF_CONTROL_PANEL)
    if(WIN32)
        target_sources(recif PRIVATE app/main_unit.c)
        target_link_libraries(recif PRIVATE user32 gdi32 comdlg32 shell32)
    else()
        # The panel drawn with SDL where the Windows one is not available.
        target_sources(recif PRIVATE app/main_unit_sdl.c)
        target_link_libraries(recif PRIVATE recif_platform recif_daf)
    endif()

    add_library(recif_terminal_ui STATIC app/terminal_ui.c)
    target_link_libraries(recif_terminal_ui PUBLIC recif_platform recif_daf)
    recif_warnings(recif_terminal_ui)

    add_executable(daf2020 app/terminal.c app/prefs.c)
    target_link_libraries(daf2020 PRIVATE recif_terminal_ui recif_transport)
    target_include_directories(daf2020 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    recif_warnings(daf2020)
    if(WIN32)
        target_sources(daf2020 PRIVATE "${RECIF_ICON_RESOURCE}")
    endif()
    if(WIN32 AND MSVC)
        target_sources(daf2020 PRIVATE app/recif.manifest)
        target_link_options(daf2020 PRIVATE /ENTRY:mainCRTStartup)
        set_target_properties(daf2020 PROPERTIES WIN32_EXECUTABLE TRUE)
    endif()
endif()
